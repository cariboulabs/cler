find_package(PkgConfig REQUIRED)

function(cler_add_source_component name)
  add_library(${name} INTERFACE)
  target_link_libraries(${name} INTERFACE cler::blocks_sources_core)
  target_link_libraries(blocks_sources INTERFACE ${name})
endfunction()

if(FFMPEG_FOUND)

  target_include_directories(blocks_source_ffmpeg INTERFACE ${FFMPEG_INCLUDE_DIRS})
  target_link_libraries(blocks_source_ffmpeg INTERFACE avformat avcodec swresample avutil m dl stdc++ pthread)
endif()

pkg_check_modules(CARIBOULITE cariboulite)

  message(STATUS "CaribouLite found: ${CARIBOULITE_LIBRARIES}")
  cler_add_source_component(blocks_source_cariboulite)
  add_library(cler::blocks_source_cariboulite ALIAS blocks_source_cariboulite)
  target_include_directories(blocks_source_cariboulite INTERFACE ${CARIBOULITE_INCLUDE_DIRS})
  target_link_libraries(blocks_source_cariboulite INTERFACE ${CARIBOULITE_LIBRARIES})

  set(CARIBOULITE_FOUND ${CARIBOULITE_FOUND} CACHE INTERNAL "CaribouLite found")

  message(WARNING "CaribouLite package not found: SourceCaribouliteBlock will not be available")
endif()
